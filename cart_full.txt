from decimal import Decimal

from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy import text
from sqlalchemy.orm import Session
from sqlalchemy.exc import IntegrityError

from app.api.deps import get_current_user
from app.db.session import get_db
from app.models.commerce import Cart, CartItem
from app.schemas.cart import (
    AddCartItemRequest,
    UpdateCartItemRequest,
    RemoveCartItemRequest,
)

router = APIRouter(prefix="/api/cart", tags=["cart"])


def _get_or_create_cart(db: Session, tenant_id: int, user_id: int):
    cart = (
        db.query(Cart)
        .filter(
            Cart.tenant_id == tenant_id,
            Cart.user_id == user_id,
            Cart.status == "ACTIVE",
        )
        .first()
    )

    if not cart:
        cart = Cart(
            tenant_id=tenant_id,
            user_id=user_id,
            status="ACTIVE",
        )
        db.add(cart)
        db.flush()

    return cart


def _get_cart_item_query(
    db: Session,
    cart_id: int,
    product_id: int,
    store_id: int,
    delivery_option_id: int,
):
    """
    Marketplace cart identity.
    Same product from different stores or with different delivery options must be
    treated as separate cart rows.
    """
    return db.query(CartItem).filter(
        CartItem.cart_id == cart_id,
        CartItem.product_id == product_id,
        CartItem.store_id == store_id,
        CartItem.delivery_option_id == delivery_option_id,
    )

def _get_store_product(
    db: Session,
    tenant_id: int,
    store_id: int,
    product_id: int,
):
    row = db.execute(
        text(
            """
            SELECT
                sp.store_id,
                sp.product_id,
                sp.stock_qty,
                sp.reserved_qty,
                sp.local_price,
                sp.is_active,
                p.product_name,
                p.image_url,
                p.base_price,
                p.sale_price,
                p.currency_code,
                p.is_active AS product_active,
                s.store_name,
                s.seller_display_name,
                s.ships_from_name,
                s.store_email,
                s.seller_contact_email
            FROM store_products sp
            JOIN products p
              ON p.id = sp.product_id
             AND p.tenant_id = sp.tenant_id
            JOIN stores s
              ON s.id = sp.store_id
             AND s.tenant_id = sp.tenant_id
            WHERE sp.tenant_id = :tenant_id
              AND sp.store_id = :store_id
              AND sp.product_id = :product_id
            LIMIT 1
            """
        ),
        {
            "tenant_id": tenant_id,
            "store_id": store_id,
            "product_id": product_id,
        },
    ).mappings().first()

    if not row or not row["is_active"] or not row["product_active"]:
        raise HTTPException(
            status_code=404,
            detail="Product not available in this store",
        )

    return row


def _get_delivery_option(
    db: Session,
    tenant_id: int,
    store_id: int,
    product_id: int,
    delivery_option_id: int,
):
    row = db.execute(
        text(
            """
            SELECT
                pdo.id AS delivery_option_id,
                pdo.delivery_fee,
                pdo.is_free,
                pdo.cutoff_time,
                dm.code AS delivery_method_code,
                dm.name AS delivery_label,
                dm.min_days,
                dm.max_days
            FROM product_delivery_options pdo
            JOIN delivery_methods dm
              ON dm.id = pdo.delivery_method_id
             AND dm.tenant_id = pdo.tenant_id
            WHERE pdo.id = :delivery_option_id
              AND pdo.tenant_id = :tenant_id
              AND pdo.store_id = :store_id
              AND pdo.product_id = :product_id
              AND pdo.is_active = 1
              AND dm.is_active = 1
            LIMIT 1
            """
        ),
        {
            "tenant_id": tenant_id,
            "store_id": store_id,
            "product_id": product_id,
            "delivery_option_id": delivery_option_id,
        },
    ).mappings().first()

    if not row:
        raise HTTPException(
            status_code=400,
            detail="Invalid delivery option for this product/store",
        )

    return row


def _eta_text(min_days, max_days):
    min_days = int(min_days or 0)
    max_days = int(max_days or 0)

    if min_days == 0 and max_days == 0:
        return "Today"

    if min_days == max_days:
        return f"{min_days} business day" if min_days == 1 else f"{min_days} business days"

    return f"{min_days}-{max_days} business days"


def _available_stock(row) -> int:
    return int(row["stock_qty"] or 0) - int(row["reserved_qty"] or 0)


def _final_price(row) -> Decimal:
    if row["local_price"] is not None:
        return Decimal(row["local_price"])

    if row["sale_price"] is not None:
        return Decimal(row["sale_price"])

    return Decimal(row["base_price"] or 0)


@router.get("")
def get_cart(
    db: Session = Depends(get_db),
    current_user=Depends(get_current_user),
):
    cart = (
        db.query(Cart)
        .filter(
            Cart.tenant_id == current_user.tenant_id,
            Cart.user_id == current_user.id,
            Cart.status == "ACTIVE",
        )
        .first()
    )

    if not cart:
        return {
            "items": [],
            "stores": [],
            "items_total": 0.0,
            "delivery_total": 0.0,
            "total_amount": 0.0,
        }

    items = (
        db.query(CartItem)
        .filter(CartItem.cart_id == cart.id)
        .order_by(CartItem.store_id.asc(), CartItem.id.asc())
        .all()
    )

    cart_items = []
    stores_map = {}

    items_total = Decimal("0.00")
    delivery_total = Decimal("0.00")

    for i in items:
        line_total = Decimal(i.line_total or 0)
        delivery_fee = Decimal(i.delivery_fee_snapshot or 0)

        items_total += line_total
        delivery_total += delivery_fee

        store_id = int(i.store_id) if i.store_id else None

        store_name = None
        sold_by = None
        ships_from = None

        if store_id:
            store_row = db.execute(
                text(
                    """
                    SELECT
                        store_name,
                        seller_display_name,
                        ships_from_name
                    FROM stores
                    WHERE id = :store_id
                      AND tenant_id = :tenant_id
                    LIMIT 1
                    """
                ),
                {
                    "tenant_id": current_user.tenant_id,
                    "store_id": store_id,
                },
            ).mappings().first()

            if store_row:
                store_name = store_row["store_name"]
                sold_by = store_row["seller_display_name"] or store_row["store_name"]
                ships_from = store_row["ships_from_name"] or store_row["store_name"]

        item_dict = {
            "cart_item_id": int(i.id),
            "product_id": int(i.product_id),
            "store_id": store_id,
            "store_name": store_name,
            "sold_by": sold_by,
            "ships_from": ships_from,
            "delivery_option_id": int(i.delivery_option_id)
            if i.delivery_option_id
            else None,
            "delivery_method_code": i.delivery_method_code,
            "delivery_label": i.delivery_label,
            "delivery_fee": float(delivery_fee),
            "delivery_eta": i.delivery_eta_snapshot,
            "product_name": i.product_name_snapshot,
            "image_url": i.product_image_snapshot,
            "unit_price": float(i.unit_price_snapshot),
            "quantity": int(i.quantity),
            "line_total": float(line_total),
        }

        cart_items.append(item_dict)

        if store_id not in stores_map:
            stores_map[store_id] = {
                "store_id": store_id,
                "store_name": store_name,
                "sold_by": sold_by,
                "ships_from": ships_from,
                "items": [],
                "items_total": Decimal("0.00"),
                "delivery_total": Decimal("0.00"),
                "store_total": Decimal("0.00"),
            }

        stores_map[store_id]["items"].append(item_dict)
        stores_map[store_id]["items_total"] += line_total
        stores_map[store_id]["delivery_total"] += delivery_fee
        stores_map[store_id]["store_total"] += line_total + delivery_fee

    stores = []
    for store in stores_map.values():
        stores.append(
            {
                **store,
                "items_total": float(store["items_total"]),
                "delivery_total": float(store["delivery_total"]),
                "store_total": float(store["store_total"]),
            }
        )

    return {
        "items": cart_items,
        "stores": stores,
        "items_total": float(items_total),
        "delivery_total": float(delivery_total),
        "total_amount": float(items_total + delivery_total),
    }


@router.post("/add")
def add_to_cart(
    payload: AddCartItemRequest,
    db: Session = Depends(get_db),
    current_user=Depends(get_current_user),
):
    if payload.quantity <= 0:
        raise HTTPException(
            status_code=400,
            detail="Quantity must be greater than zero",
        )

    cart = _get_or_create_cart(db, current_user.tenant_id, current_user.id)

    store_product = _get_store_product(
        db=db,
        tenant_id=current_user.tenant_id,
        store_id=payload.store_id,
        product_id=payload.product_id,
    )

    delivery_option = _get_delivery_option(
        db=db,
        tenant_id=current_user.tenant_id,
        store_id=payload.store_id,
        product_id=payload.product_id,
        delivery_option_id=payload.delivery_option_id,
    )

    price = _final_price(store_product)
    delivery_fee = Decimal(delivery_option["delivery_fee"] or 0)
    delivery_eta = _eta_text(
        delivery_option["min_days"],
        delivery_option["max_days"],
    )

    item = _get_cart_item_query(
        db=db,
        cart_id=cart.id,
        product_id=payload.product_id,
        store_id=payload.store_id,
        delivery_option_id=payload.delivery_option_id,
    ).first()

    existing_qty = int(item.quantity) if item else 0
    new_qty = existing_qty + payload.quantity

    available = _available_stock(store_product)

    if new_qty > available:
        raise HTTPException(
            status_code=400,
            detail=f"Only {available} available in this store",
        )

    if item:
        item.quantity = new_qty
        item.unit_price_snapshot = price
        item.delivery_fee_snapshot = delivery_fee
        item.delivery_eta_snapshot = delivery_eta
        item.line_total = Decimal(new_qty) * price
    else:
        item = CartItem(
            cart_id=cart.id,
            tenant_id=current_user.tenant_id,
            user_id=current_user.id,
            product_id=payload.product_id,
            store_id=payload.store_id,
            delivery_option_id=payload.delivery_option_id,
            delivery_method_code=delivery_option["delivery_method_code"],
            delivery_label=delivery_option["delivery_label"],
            delivery_fee_snapshot=delivery_fee,
            delivery_eta_snapshot=delivery_eta,
            product_name_snapshot=store_product["product_name"],
            product_image_snapshot=store_product["image_url"],
            unit_price_snapshot=price,
            quantity=payload.quantity,
            line_total=Decimal(payload.quantity) * price,
        )
        db.add(item)

    try:
        db.commit()
    except IntegrityError:
        db.rollback()

        # Race-safe fallback: another request may have inserted the exact same
        # cart identity between our SELECT and INSERT. Re-read the exact
        # marketplace cart row and update quantity.
        item = _get_cart_item_query(
            db=db,
            cart_id=cart.id,
            product_id=payload.product_id,
            store_id=payload.store_id,
            delivery_option_id=payload.delivery_option_id,
        ).first()

        if not item:
            raise HTTPException(
                status_code=500,
                detail="Unable to add item to cart. Please try again.",
            )

        existing_qty = int(item.quantity or 0)
        new_qty = existing_qty + payload.quantity

        available = _available_stock(store_product)
        if new_qty > available:
            raise HTTPException(
                status_code=400,
                detail=f"Only {available} available in this store",
            )

        item.quantity = new_qty
        item.unit_price_snapshot = price
        item.delivery_fee_snapshot = delivery_fee
        item.delivery_eta_snapshot = delivery_eta
        item.line_total = Decimal(new_qty) * price
        db.commit()

    return {
        "message": "Added to cart",
        "product_id": payload.product_id,
        "store_id": payload.store_id,
        "delivery_option_id": payload.delivery_option_id,
        "quantity": new_qty,
        "unit_price": float(price),
        "delivery_fee": float(delivery_fee),
        "delivery_eta": delivery_eta,
    }


@router.post("/update")
def update_cart(
    payload: UpdateCartItemRequest,
    db: Session = Depends(get_db),
    current_user=Depends(get_current_user),
):
    cart = _get_or_create_cart(db, current_user.tenant_id, current_user.id)

    if payload.store_id is None or payload.delivery_option_id is None:
        raise HTTPException(
            status_code=400,
            detail="store_id and delivery_option_id are required for marketplace cart updates",
        )

    item = _get_cart_item_query(
        db=db,
        cart_id=cart.id,
        product_id=payload.product_id,
        store_id=payload.store_id,
        delivery_option_id=payload.delivery_option_id,
    ).first()

    if not item:
        raise HTTPException(status_code=404, detail="Cart item not found")

    if payload.quantity <= 0:
        db.delete(item)
        db.commit()
        return {"message": "Cart item removed"}

    if not item.store_id:
        raise HTTPException(status_code=400, detail="Cart item has no store_id")

    store_product = _get_store_product(
        db=db,
        tenant_id=current_user.tenant_id,
        store_id=item.store_id,
        product_id=payload.product_id,
    )

    available = _available_stock(store_product)

    if payload.quantity > available:
        raise HTTPException(
            status_code=400,
            detail=f"Only {available} available in this store",
        )

    price = _final_price(store_product)

    item.quantity = payload.quantity
    item.unit_price_snapshot = price
    item.line_total = Decimal(payload.quantity) * price

    db.commit()

    return {
        "message": "Cart updated",
        "product_id": payload.product_id,
        "store_id": int(item.store_id),
        "delivery_option_id": int(item.delivery_option_id)
        if item.delivery_option_id
        else None,
        "quantity": payload.quantity,
    }


@router.post("/remove")
def remove_cart(
    payload: RemoveCartItemRequest,
    db: Session = Depends(get_db),
    current_user=Depends(get_current_user),
):
    cart = _get_or_create_cart(db, current_user.tenant_id, current_user.id)

    if payload.store_id is None or payload.delivery_option_id is None:
        raise HTTPException(
            status_code=400,
            detail="store_id and delivery_option_id are required for marketplace cart updates",
        )

    item = _get_cart_item_query(
        db=db,
        cart_id=cart.id,
        product_id=payload.product_id,
        store_id=payload.store_id,
        delivery_option_id=payload.delivery_option_id,
    ).first()

    if item:
        db.delete(item)
        db.commit()

    return {"message": "Removed from cart"}
